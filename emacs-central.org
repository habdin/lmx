* What is emacs-central.org?
#+BEGIN_SRC org :tangle README.org
  Emacs is an advanced editor with enormous capabilities, The fact that it is built with Emacs-lisp
  that is a dialect of LISP programming language makes it a powerful editor. Natively, Emacs is built
  with defaults that suited its origins. The current builtin Emacs tools are capable yet they lag
  behind current other editors and IDEs. Luckily, Emacs LISP can turn Emacs itself into a modern
  looking text editor with many features that both novice computer user and experienced programmer
  would appreciate. Emacs is highly configurable and customizable through the use of Emacs
  LISP. Configuring Emacs properly can turn it into this powerful editor.

  Emacs configuration currently resides either in a .emacs file in the user $HOME dir, in an "init.el"
  file within a ".emacs.d" dir under the user $HOME dir or in an "init.el" file that resides within an
  "emacs" directory within $XDG-CONFIG-DIR. If a user uses a single file to write down many tweaks and
  configuration, this file might soon becomes unorganized. Using Org-mode file(s) to configure Emacs
  become handy when the user needs an "organized" complex Emacs configuration.

  Within Light Emacs, Emacs configuration is moulded in a Central Emacs Org-mode file that is called
  "emacs-central.org". The aim is that if a user wants to use Light Emacs, he (she) can regenerate the
  whole sibling configuration files by simply tangling this file.

#+END_SRC

* LMX Core components

Light Emacs distribution core components include the following ones:

1- Emacs initialization files (early-init.el and init.el)
2- Distribution core file
3- Distribution user interface file
4- Distribution library file
5- Distribution package management file

** LMX Initialization

*** Emacs Early Init file

Many of the settings put within this section are put after reading "Elisp
info>Init file". The aim of the statements is to remove unnecessary programs
from /emacs initialization/ to boost the latter.

#+BEGIN_SRC emacs-lisp :tangle ~/.config/emacs/early-init.el
  ;;; early-init.el -*- lexical-binding: t -*-

  ;; Postpone garbage collection till after emacs is initiated
  (setq gc-cons-threshold most-positive-fixnum)

  ;; Distribution default language settings
  (set-language-environment "utf-8")
  (set-default-coding-systems 'utf-8)

  ;; Disable package.el initialization at emacs startup
  (setq package-enable-at-startup nil)
  (advice-add #'package--ensure-init-file :override #'ignore)

  ;; Set the emacs distribution project variable defaults
  ;; These are defined here to ease the process loading lmx distributin lisp codebase
  (defconst emacs-distrib-proj-root (expand-file-name "lmx" "~/Documents/Projects")
    "Lmx emacs distribution project root")
  (defconst distrib-lisp-code-dir (expand-file-name "lisp" emacs-distrib-proj-root)
    "Lmx emacs lisp codebase directory")

  ;; Set the eln-cache path
  ;; The purpose of this sexp is to burry the eln-cache inside a .cache within the proj root dir
  (startup-redirect-eln-cache (file-name-as-directory
    			     (expand-file-name ".cache/eln-cache" emacs-distrib-proj-root)))
  (setq native-comp-deferred-compilation nil)

  ;; Always prefer newer .el files over .elc
  (setq load-prefer-newer t)

  ;; Add the distribution lisp codebase to load-path
  (let ((default-directory (eval distrib-lisp-code-dir)))
    (normal-top-level-add-subdirs-to-load-path))
  ;; (add-to-list 'load-path (expand-file-name "lisp/core" emacs-distrib-proj-root))
  ;; Load the distribution lisp codebase
  (require 'distrib-lib)

  ;; Disable menu-bar explicitly. Disable tool-bar-mode and scroll-bar-mode when these
  ;; features exist in emacs.
  (menu-bar-mode 0)
  (when (fboundp 'tool-bar-mode)
    (tool-bar-mode 0))
  (when (fboundp 'scroll-bar-mode)
    (scroll-bar-mode 0))
  (setq frame-inhibit-implied-resize t)

  ;; Disable the splashnnn screen, Set the initial major mode, clear the scratch buffer message.
  (setq inhibit-splash-screen t
        initial-scratch-message nil
        initial-major-mode #'fundamental-mode)

  ;; Performance tuning variables and environment variables that enhance lsp-mode functionality
  (setq read-process-output-max (* 1024 1024))
  (setenv "LSP_USE_PLISTS" "true")
  (setq max-lisp-eval-depth 48000)

**** Distribution package management dependency file

The package management dependency file configures the *package installer*,
and other package list software.

#+BEGIN_SRC emacs-lisp :mkdirp yes :tangle lisp/core/distrib-packagers.el
  ;;; distrib-packagers.el -*- lexical-binding: t; -*-

  (require 'distrib-lib)

  ;; Initialize Straight package manager
  (setq straight-base-dir distrib-local-dir
  	straight-repository-branch "develop"
  	straight-cache-autoloads t
  	straight-check-for-modifications nil
  	straight-enable-package-integration nil
  	straight-vc-git-default-clone-depth 1
  	autoload-compute-prefixes nil
  	straight-use-package-by-default t)

  (defvar bootstrap-version)
  (setq straight-built-in-pseudo-packages '(project python))
  (let ((bootstrap-file
  	 (expand-file-name "straight/repos/straight.el/bootstrap.el" distrib-local-dir))
  	(bootstrap-version 5))
    (unless (file-exists-p bootstrap-file)
      (with-current-buffer
  	(url-retrieve-synchronously
  	 "https://raw.githubusercontent.com/raxod502/straight.el/develop/install.el"
  	 'silent 'inhibit-cookies)
  	(goto-char (point-max))
  	(eval-print-last-sexp)))
    (load bootstrap-file nil 'nomessage))

  ;; Install use-package in emacs version prior to 29.1
  (when (version< emacs-version "29.1")
    (straight-use-package 'use-package)
    (require 'use-package))

  ;; Set use-package macro to always use defer.
  ;; Note that whenever you need to ensure you should explicitly add it for any relevant package.
  (use-package use-package
    :init
    (setq use-package-always-defer t))

  ;; Configure Package

  (use-package package
    :config
    (setq package-user-dir (expand-file-name "elpa" distrib-local-dir)
  	package-gnupghome-dir (expand-file-name "gpg" package-user-dir))
    (add-to-list 'package-archives
  	       '("gnu" . "https://elpa.gnu.org/packages/") t)
    (add-to-list 'package-archives
  	       '("melpa" . "https://melpa.org/packages/") t)
    (add-to-list 'package-archives
  	       '("org" . "https://orgmode.org/elpa/") t))

  ;; Install Paradox for browsing packages
  (use-package paradox
    :straight t
    :config
    (setq paradox-github-token (expand-file-name "paradox-token" user-emacs-directory)))


  ;; Install diminish and delight for use with use-package when needed.
  (use-package diminish
    :straight t
    )

  (use-package delight
    :straight t
    )


  ;; Install epkg to view packages in Emacsmirror
  (use-package epkg
    :straight t
    :config
    (setq epkg-repository (expand-file-name "epkgs" distrib-local-dir)))

  (provide 'distrib-packagers)

(defun distrib-evil-exit-insert ()
  "Exit evil insert mode when 'jk' key-chord is pressed"
  (interactive)
  (let ((event (read-event nil nil 0.3)))
	(if (and (characterp event) (char-equal event ?k))
		(evil-normal-state)
	  (insert "j")
	  (push event unread-command-events))))

(define-key evil-insert-state-map (kbd "j") 'distrib-evil-exit-insert)
